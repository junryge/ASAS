1. 시스템 아키텍처

┌──────────┐   SECS/GEM   ┌──────────────┐   UDP/IP    ┌───────────────┐
│   HOST   │◄────────────►│   FabScope   │◄──────────►│  CLW MCP7     │
│ (MES/ERP)│              │ (SysView+MSS)│  3600/3500  │  (AGV 제어)   │
└──────────┘              └──────┬───────┘            └───────┬───────┘
                                 │                            │
                           UDP/IP│                     TCP/IP │ AMP Remote.DLL
                                 │                            │
                          ┌──────▼───────┐            ┌───────▼───────┐
                          │ Atlas 서버   │            │   AGV 설비    │
                          │ (UDP 수신)   │            │  프로그램     │
                          └──────────────┘            └───────────────┘

SK Hynix Fab 환경의 AGV 시스템은 HOST(MES/ERP)에서 FabScope로 지시가 전달되고, CLW MCP7을 통해 UDP/IP로 Atlas 서버에 실시간 상태가 보고됩니다. 동시에 AGV 설비는 AMP Remote.DLL을 통해 HOST와 직접 통신하여 제어됩니다.

2.3 UDP 패킷 파싱 방법

수신된 UDP 패킷은 CSV 형태이므로 콤마로 분리하여 파싱합니다.

수신 패킷 예시: 2,OHT,V047,1,1,0000,1,1232,0,1202,4,4,PIN2702,20308,00000000,0000,IR5S005S_2,STB031-R08,50,0,0
파싱 방법: Split(',') → fields[0]=텍스트ID, fields[1]=MCP명칭, fields[2]=Vehicle명, ...

7. Atlas Server — Java UDP 수신 및 Logpresso 저장

개요: CLW MCP7에서 전송되는 UDP 패킷을 Atlas 서버에서 수신하여 Vehicle/Rail Edge 상태를 분석하고, 이를 Logpresso 데이터베이스에 저장하는 실시간 데이터 파이프라인입니다.

7.1 OhtUdpListener — UDP 소켓 수신

Atlas 서버는 UDP 포트 3600 (SysView) 또는 3500 (MSS)에서 MCP 메시지를 수신합니다.

public class OhtUdpListener implements Runnable {
    private DatagramSocket socket;
    private byte[] buffer = new byte[1500];

    public OhtUdpListener(int port) throws SocketException {
        this.socket = new DatagramSocket(port);
    }

    @Override
    public void run() {
        while (!Thread.currentThread().isInterrupted()) {
            try {
                DatagramPacket packet = new DatagramPacket(buffer, buffer.length);
                socket.receive(packet);

                String message = new String(packet.getData(), 0, packet.getLength()).trim();

                // DataService 큐에 추가 → 비동기 처리
                Msg msg = new Msg();
                msg.setFabId("FAB");
                msg.setMcpName("CLW_MCP7");
                msg.setMsg(message);
                msg.setTimestamp(System.currentTimeMillis());

                DataService.getInstance().getQueue().add(msg);
            } catch (IOException e) {
                logger.error("UDP receive error", e);
            }
        }
        socket.close();
    }
}

7.2 메시지 파싱 — OhtMsgWorkerRunnable

수신된 CSV 형식의 메시지를 콤마로 분리하여 Text ID별로 라우팅합니다.

public class OhtMsgWorkerRunnable implements Runnable {
    private Msg msg;

    @Override
    public void run() {
        String[] tokens = StringUtils.splitPreserveAllTokens(msg.getMsg(), ',');

        if (tokens.length < 2) return;

        String messageId = tokens[0];  // Text ID
        String mcpName = tokens[1];    // MCP 명칭

        try {
            switch (messageId) {
                case "1":
                    _processMcpOnline(tokens, msg);
                    break;
                case "2":
                    _processOhtReport(tokens, msg);  // Vehicle 상태
                    break;
                case "3":
                    _processAbnormalReport(tokens, msg);
                    break;
                case "4":
                    _processDeviceStatus(tokens, msg);
                    break;
                case "15":
                    _processPathReport(tokens, msg);
                    break;
                case "101":
                    _processTscStatus(tokens, msg);  // TSC 상태
                    break;
            }
        } catch (Exception e) {
            logger.error("Message parsing error: {}", msg.getMsg(), e);
        }
    }

    private void _processOhtReport(String[] tokens, Msg msg) {

        int currentPosition = Integer.parseInt(tokens[7]);

        if (edge != null && state == 1) {  // 1 = 운전중
            // 속도 업데이트
        }
    }
}

7.3 RailEdge 속도 업데이트

차량이 주행 중인 Rail Edge의 속도를 가중평균으로 누적하여 실시간 트래픽 분석에 반영합니다.

public class RailEdge {
    private String id;
    private double velocity = 0.0;           // 평균 속도 (m/s)
    private double maxVelocity = 2.0;        // 최대 속도 설정값
    private boolean changedVelocity = false; // 변경 플래그
    private int hisCnt = 0;                  // 누적 카운트

    public void addVelocity(double velocity) {
        // NaN/Inf 체크
        if (Double.isNaN(velocity) || Double.isInfinite(velocity)) {
            return;
        }

        // 속도 범위 제한 (1.5 ~ maxVelocity m/s)
        velocity = Math.max(1.5, Math.min(velocity, maxVelocity));

        // 가중평균 계산
        if (hisCnt > 0) {
            double weight = PredictionPara.getInstance().getLastHisWeight();
            // 기존 속도에 더 큰 가중치, 신규 속도는 보정
            this.velocity = (this.velocity * weight) + (velocity * (1.0 - weight));
        } else {
            this.velocity = velocity;  // 초기값
        }

        this.changedVelocity = true;
        hisCnt++;
    }

    public double getVelocity() {
        return velocity;
    }

    public String getId() {
        return id;
    }
}

7.5 TrafficBatch — 1분 배치 작업

매 1분마다 모든 Rail Edge의 현재 속도/트래픽 상태를 스냅샷으로 저장하는 배치 작업입니다.

public class TrafficBatch implements Job {

    @Override
    public void execute(JobExecutionContext context) throws JobExecutionException {
        List logpressoData = new ArrayList<>();

        // 모든 RailEdge 순회
        List<RailEdge> railEdges = RailEdgeService.getInstance().getAllEdges();

        for (RailEdge edge : railEdges) {
            Tuple t = new Tuple();

            // 기본 정보
            t.put("createTime", System.currentTimeMillis());
            t.put("fabId", "FAB");
            t.put("railEdgeId", edge.getId());

            // 트래픽 데이터
            t.put("velocity", edge.getVelocity());
            t.put("maxVelocity", edge.getMaxVelocity());
            t.put("absoluteVelocity", edge.getAbsoluteVelocity());

            // 차량/통행 통계
            t.put("vhlCnt", edge.getVehicleCount());        // 현재 주행 차량 수
            t.put("passCnt", edge.getTotalPassCount());     // 누적 통행 수

            // 위치 정보
            t.put("HID_ID", edge.getHIDId());

            logpressoData.add(t);
        }

        // Logpresso 배치 저장
        if (!logpressoData.isEmpty()) {
            Util.insertInLogpressoDatabase(logpressoData, "ATLAS_RAIL_TRAFFIC", "TrafficBatch");
        }
    }
}

// Quartz 스케줄러 등록 (1분 주기)

ATLAS_RAIL_TRAFFIC (1분 주기 Rail 트래픽)

필드명 타입 설명

Logpresso API 래퍼

public class LogpressoAPI {

    /**
     * Tuple을 Logpresso 테이블에 저장
     * @param tableName 테이블명 (예: "ATLAS_OHT_HID_OFF")
     * @param tuples 저장할 Tuple 리스트
     * @param timeoutSeconds 저장 타임아웃 (초)
     */
    public static void setInsertTuples(String tableName, List<Tuple> tuples,
                                       int timeoutSeconds) {
        try {
            // Logpresso JDBC 커넥션
            // INSERT INTO tableName ...
            logger.info("Inserted {} records into {}", tuples.size(), tableName);
        } catch (Exception e) {
            logger.error("Insert failed: {}", tableName, e);
        }
    }
}

public class Util {

    /**
     * Util 래퍼: 로깅과 함께 저장
     */
    public static void insertInLogpressoDatabase(List<Tuple> tuples,
                                                  String tableName, String caller) {
        if (tuples == null || tuples.isEmpty()) {
            logger.warn("[{}] No data to insert into {}", caller, tableName);
            return;
        }

        try {
            LogpressoAPI.setInsertTuples(tableName, tuples, 20);
            logger.info("[{}] {} records inserted into {}", caller, tuples.size(), tableName);
        } catch (Exception e) {
            logger.error("[{}] Logpresso insert failed for {}", caller, tableName, e);
        }
    }
}

Logpresso SQL 쿼리 예시

-- 1. 최근 1시간 Rail Edge별 평균 속도
SELECT railEdgeId, AVG(velocity) as avg_velocity, AVG(vhlCnt) as avg_vehicles
FROM ATLAS_RAIL_TRAFFIC
WHERE createTime >= now() - 3600000
GROUP BY railEdgeId
ORDER BY avg_velocity ASC;

-- 2. Vehicle V047의 HID 수신 기록 (최근 24시간)
SELECT MCP_NM, VHL_ID, HID_ID, FROM_ADDRESS, TO_ADDRESS,
       FROM_UNIXTIME(OFF_TIME/1000) as off_time
FROM ATLAS_OHT_HID_OFF
WHERE VHL_ID = 'V047' AND OFF_TIME >= now() - 86400000
ORDER BY OFF_TIME DESC;

-- 3. 가장 혼잡한 Rail Edge (실시간)
SELECT railEdgeId, velocity, vhlCnt, passCnt
FROM ATLAS_RAIL_TRAFFIC
WHERE createTime = (SELECT MAX(createTime) FROM ATLAS_RAIL_TRAFFIC)
ORDER BY vhlCnt DESC LIMIT 10;

-- 4. HID별 일일 통과 차량 수
SELECT HID_ID, COUNT(DISTINCT VHL_ID) as unique_vehicles, COUNT(*) as total_passes
FROM ATLAS_OHT_HID_OFF
WHERE DATE(FROM_UNIXTIME(OFF_TIME/1000)) = CURDATE()
GROUP BY HID_ID;
